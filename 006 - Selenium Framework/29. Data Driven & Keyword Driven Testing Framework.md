## Data Driven Testing Framework (DDTF)
* DDTF is a kind of automation testing framework which separates the Test Data from the Test Scripts.
* With the help of DDTF we can able to create multiple test cases and run those test cases with multiple dataset at once.
* Always use DDTF along with POM only.
* The test data is kept in an external files like:- 
1. Excel file
2. CSV file 
3. XML file
4. MySQL databases
5. JSON file
6. NoSQL databases

## Advantages of DDTF
* Multiple test cases with multiple datasets can be run at the same time
* Large volume of test data can be executed easily
* Easy to maintain
* Reusable code

## Excel Functions
~~~python
"""
ExcelFunctions.py


This is a Python Class file to Read and Write an Excel file
"""


from openpyxl import load_workbook


class ExcelFunctions:


   def __init__(self, file_name, sheet_name):
       self.file = file_name
       self.sheet = sheet_name


   # Fetch the total row count from Excel sheet
   def row_count(self):
       workbook = load_workbook(self.file)
       sheet = workbook[self.sheet]
       return sheet.max_row


   # Fetch the total column count from the Excel sheet
   def column_count(self):
       workbook = load_workbook(self.file)
       sheet = workbook[self.sheet]
       return sheet.max_column


   # Read the data from Excel sheet of specific Row and Column
   def read_data(self, row_number, column_number):
       workbook = load_workbook(self.file)
       sheet = workbook[self.sheet]
       return sheet.cell(row=row_number, column=column_number).value


   # Write the data into an Excel sheet on a specific Row and Column
   def write_data(self, row_number, column_number, data):
       workbook = load_workbook(self.file)
       sheet = workbook[self.sheet]
       sheet.cell(row=row_number, column=column_number).value = data
       workbook.save(self.file)

~~~

## test_login.py
~~~python
from selenium import webdriver
from selenium.webdriver.chrome.service import Service
from selenium.webdriver.common.by import By
from webdriver_manager.chrome import ChromeDriverManager
from TestLocators.locators import HRMLocators
from TestData.data import HRMData
from Utilities.excel_functions import ExcelFunctions


class Test_SauceDDTF:

    def test_login_excel(self):
        self.excel_file = HRMData().excel_file
        sheet_number = HRMData().sheet_number

        self.excel = ExcelFunctions(self.excel_file, sheet_number)
        self.driver = webdriver.Chrome(service=Service(ChromeDriverManager().install()))

        self.driver.get(HRMData().url)
        self.driver.maximize_window()

        self.driver.implicitly_wait(10)

        self.rows = self.excel.row_count()

        for row in range(2, self.rows+1):
            username = self.excel.read_data(row, 5)
            password = self.excel.read_data(row, 6)


            self.driver.find_element(by=By.NAME, value=HRMLocators().username_locator).send_keys(username)
            self.driver.find_element(by=By.NAME, value=HRMLocators().password_locator).send_keys(password)
            self.driver.find_element(by=By.ID, value=HRMLocators().submit_button).click()


            # Main Validation of the TEST CASE either PASS or FAIL is going to happen
            if HRMData().dashboard_url in self.driver.current_url:
                print("SUCCESS : Login success")
                self.excel.write_data(row, 7, "TEST PASS")
                self.driver.back()

            elif HRMData().url in self.driver.current_url:
                print("FAIL : Login failed")
                self.excel.write_data(row, 7, "TEST FAIL")
                self.driver.refresh()

        self.driver.quit()

~~~

## Keyword Driven Testing Framework (KDTF)

* In KDTF, we get the test-data in form of Key-value pairs
* So, the data can be saved either inside a 
1. YAML file
2. JSON file


* The data we fetch is in form of a Python dictionary
* YAML is a configuration file
* Everything is kept in a KEY-VALUE pair and being read from it
* Install using "pip install pyyaml"

## config.yaml file
~~~python

username : 'standard_user'
~~~

## yaml functions
~~~python
"""
yaml_functions.py


Python file which has YAML functions
"""

from yaml import load
from yaml.loader import SafeLoader


class YAML_Reader:


   def __init__(self, file_name):
       self.file = file_name


   def reader(self):
       with open(self.file) as file:
           data = load(file, Loader=SafeLoader)
           return data

~~~

## test_sauce_login
~~~python
"""
OrangeHRM.py

Main Execution file
"""
from Utilities.yaml_functions import YAML_Reader
from selenium import webdriver
from selenium.webdriver.chrome.service import Service
from selenium.webdriver.common.by import By
from webdriver_manager.chrome import ChromeDriverManager
from selenium.common.exceptions import NoSuchElementException
from selenium.common.exceptions import ElementNotVisibleException


class Test_SauceYAML:

   def test_saucelogin(self):

      file_name = "D:\\Automation\\samp1\\KDTF\\TestData\\config.yaml"

      yaml_reader = YAML_Reader(file_name)

      driver = webdriver.Chrome(service=Service(ChromeDriverManager().install()))

      try:
         driver.maximize_window()
         driver.get(yaml_reader.reader()['url'])
         driver.implicitly_wait(10)
         driver.find_element(by=By.NAME, value=yaml_reader.reader()['username_locator']).send_keys(yaml_reader.reader()['username'])
         driver.find_element(by=By.NAME, value=yaml_reader.reader()['password_locator']).send_keys(yaml_reader.reader()['password'])
         driver.find_element(by=By.ID, value=yaml_reader.reader()['submit_button']).click()
         
      except (NoSuchElementException, ElementNotVisibleException) as error:
         print(error)

      finally:
         driver.quit()
~~~
