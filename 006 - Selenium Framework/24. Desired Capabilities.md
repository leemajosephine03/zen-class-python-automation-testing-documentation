### **Introduction to Browser Capabilities**  
* Browser capabilities define the features, behaviors, and settings of a web browser during automated testing.
* They help configure browser properties such as window size, headless mode, browser version, and other preferences.  

### **Chrome Desired Capabilities**  
Desired capabilities for Chrome are used to set properties before launching a Chrome browser instance in Selenium.  
- Defined using `ChromeOptions`.  
- Common capabilities:  
  - `add_argument("--headless")` → Run Chrome without a UI.  
  - `add_argument("--disable-gpu")` → Disable GPU usage.  
  - `add_experimental_option("prefs", prefs)` → Set browser preferences.  
  - `add_argument("--incognito")` → Open Chrome in incognito mode.  

## 1. Automating Chrome Browser with configured window size

```python
"""
Open the chrome browser in 640 x 450 configuration
Don't use service object XXX
"""
from selenium import webdriver
from selenium.webdriver.chrome.options import Options
from selenium.webdriver.chrome.service import Service
from webdriver_manager.chrome import ChromeDriverManager
from selenium.webdriver.common.by import By
from time import sleep


# Create Data Class to store all the data like username, password, url
class Data:
    url = "https://opensource-demo.orangehrmlive.com/web/index.php/auth/login"
    username = "Admin"
    password = "admin123"


# Create Locators Class to store all the locators like xpath, id, class
class Locators:
    username_locator = 'username'
    password_locator = 'password'
    login_button_locator = '//*[@id="app"]/div[1]/div/div[1]/div/div[2]/div[2]/form/div[3]/button'


# Create the WindowConfiguration Class to define the methods
class WindowConfiguration(Data, Locators):

    def window_size_automation(self, width, height):
        try:
            chrome_options = Options()
            chrome_options.add_argument(f"--window-size={width},{height}")
            driver = webdriver.Chrome(options=chrome_options)
            driver.get(self.url)
            sleep(5)
            driver.find_element(by=By.NAME, value=self.username_locator).send_keys(self.username)
            sleep(2)
            driver.find_element(by=By.NAME, value=self.password_locator).send_keys(self.password)
            sleep(2)
            driver.find_element(by=By.XPATH, value=self.login_button_locator).click()
            sleep(5)
            window_size = driver.get_window_size()
            print("Window Width", window_size['width'])
            print("Window Height", window_size['height'])
            print("SUCCESS, Window launched with the specified configuration !")
        except Exception as error:
            print("ERROR", error)


# create an object and call the methods
myCapability = WindowConfiguration()
myCapability.window_size_automation(640, 450)
```

### **Firefox Desired Capabilities**  
Similar to Chrome, Firefox has capabilities that configure browser settings.  
- Defined using `FirefoxOptions`.  
- Common capabilities:  
  - `add_argument("-headless")` → Run Firefox in headless mode.  
  - `set_preference("browser.download.dir", "/path")` → Set download directory.  
  - `set_preference("privacy.trackingprotection.enabled", True)` → Enable tracking protection.

## 5. Automate Login using Cookies only

```python
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.chrome.service import Service
from webdriver_manager.chrome import ChromeDriverManager
from selenium.webdriver.common.keys import Keys
import time
import json


# Create Data Class to store all the data like username, password, url
class Data:
    url = "https://opensource-demo.orangehrmlive.com/web/index.php/auth/login"
    username = "Admin"
    password = "admin123"

# Create Locators Class to store all the locators like xpath, id, class
class Locators:
    username_locator = 'username'
    password_locator = 'password'
    login_button_locator = '//*[@id="app"]/div[1]/div/div[1]/div/div[2]/div[2]/form/div[3]/button'

# Create the WindowConfiguration Class to define the methods
class CookiesLogin(Data, Locators):

    def __init__(self):
        self.driver = webdriver.Chrome(service=Service(ChromeDriverManager().install()))

    # Function to log in and save cookies
    def login_and_save_cookies(self):
        try:
            self.driver.get("https://www.saucedemo.com/")  
            
            # Locate login elements and enter credentials
            self.driver.find_element(By.ID, "user-name").send_keys("standard_user")
            self.driver.find_element(By.ID, "password").send_keys("secret_sauce")
            self.driver.find_element(By.ID, "password").send_keys(Keys.RETURN)  # Press Enter
            
            time.sleep(5)  # Wait for login to complete
            
            # Save cookies
            cookies = self.driver.get_cookies()
            with open("cookies.json", "w") as file:
                json.dump(cookies, file)
            
            print("Cookies saved successfully!")
        except Exception as error:
            print("ERROR", error) 
        finally:
            self.driver.quit()  

    # Function to load cookies and login using them
    def login_using_cookies(self):
        try:
            self.driver.get("https://www.saucedemo.com/")  # Replace with the target website

            # Load cookies
            with open("cookies.json", "r") as file:
                cookies = json.load(file)
                for cookie in cookies:
                    self.driver.add_cookie(cookie)

            time.sleep(5)
            # Check Inventory page
            self.driver.get("https://www.saucedemo.com/inventory.html") 
            time.sleep(5)  # Wait for session to load
            
            print("Logged in using cookies!")
            self.driver.quit()
        except Exception as error:
            print("ERROR", error)            
        finally:
            self.driver.quit() 

if __name__ == "__main__":
    my_cookies =  CookiesLogin()
    # uncomment to run
    # my_cookies.login_and_save_cookies()  # Run this first to save cookies
    my_cookies.login_using_cookies()     # Run this next to log in using cookies
```
