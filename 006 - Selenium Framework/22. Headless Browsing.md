Headless Browser & Navigation
## Definition
* Headless generally means an object/ thing with no head, and in context to browsers, it means browser simulation, which has no UI.
* Headless browser automation uses a web browser for end-to-end testing without loading the browser’s UI.

## Why use a headless browser for test execution?
* Beneficial in web scraping
* Support for multiple browser versions
* Useful in CI pipelineFaster automation test execution
* Multi-Tasking

## Limitations
* Debugging will not be feasible
* Headless browsers don't mimic the exact user behavior
* The location of a web element, the color of a web element may get missed
  
## Headless Browsing on Firefox Browser
~~~python
"""
Python Selenium Headless Browsing-Firefox Browser
"""
from selenium import webdriver
from selenium.webdriver.firefox.service import Service
from webdriver_manager.firefox import GeckoDriverManager
from time import sleep


class SumanHeadlessBrowsing:


   def __init__(self, web_url):
      
       self.url = web_url


       # Code for headless browsing in Firefox
       firefox_options = webdriver.FirefoxOptions()
       firefox_options.add_argument('--headless')


       # Firefox webdriver
       self.driver = webdriver.Firefox(service=Service(GeckoDriverManager().install()), options=firefox_options)


   def start(self):
       try:
           self.driver.get(self.url)
           sleep(3)
           print("PROCESS-1 : Firefox Browser is working")
           return True
       except Exception as error:
           print("ERROR : Automation Failed to Start !")
           return False


   def shutdown(self):
       print("Process-2 : Automation closed")
       self.driver.close()


   def fetch_url(self):
       if self.start():
           return self.driver.current_url
       else:
           return False


   def fetch_title(self):
       if self.start():
           return self.driver.title
       else:
           return False


# main execution function


if __name__ == "__main__":


   url = "https://www.guvi.in/courses/"
   suman = SumanHeadlessBrowsing(url)
   suman.start()
   print(suman.fetch_url())
   print(suman.fetch_title())
   suman.shutdown()
~~~

## Headless Automation Selenium Codes
* HeadlessBrowsing.py file contains all methods
~~~python
"""
HeadlessBrowsing.py


Python Selenium automation codes
"""
from selenium import webdriver
from selenium.webdriver.chrome.service import Service
from webdriver_manager.chrome import ChromeDriverManager
from time import sleep
from selenium.webdriver.common.by import By


class Data:
   url = "https://opensource-demo.orangehrmlive.com/web/index.php/auth/login"
   dashboard_url = "https://opensource-demo.orangehrmlive.com/web/index.php/dashboard/index"
   username = "Admin"
   password = "admin123"


class Locators:
   username_input_box = "username"
   password_input_box = "password"
   submit_button = '//*[@id="app"]/div[1]/div/div[1]/div/div[2]/div[2]/form/div[3]/button'




class SumanHeadlessBrowsing(Data, Locators):


   def __init__(self, web_url):
       self.url = web_url


       # enable headless browsing in Google Chrome
       chrome_options = webdriver.ChromeOptions()
       chrome_options.add_argument('--headless')


       # webdriver with headless browsing
       self.driver = webdriver.Chrome(service=Service(ChromeDriverManager().install()), options=chrome_options)


   def start(self):
       try:
           self.driver.get(self.url)
           sleep(3)
           return True
       except Exception as error:
           print("ERROR : ", error)
           return False


   def shutdown(self):
       self.driver.close()
       return None


   def validate_username_input_box(self):
       username_input_box = self.driver.find_element(by=By.NAME, value=self.username_input_box)
       if username_input_box.is_displayed():
           return True
       else:
           return False


   def validate_password_input_box(self):
       password_input_box = self.driver.find_element(by=By.NAME, value=self.password_input_box)
       if password_input_box.is_displayed():
           return True
       else:
           return False


   def validate_submit_button(self):
       submit_button = self.driver.find_element(by=By.XPATH, value=self.submit_button)
       if submit_button.is_enabled():
           return True
       else:
           return False


   def validate_login(self):
       try:
           self.driver.find_element(by=By.NAME, value=self.username_input_box).send_keys(self.username)
           sleep(2)
           self.driver.find_element(by=By.NAME, value=self.password_input_box).send_keys(self.password)
           sleep(2)
           self.driver.find_element(by=By.XPATH, value=self.submit_button).click()
           if self.driver.current_url == self.dashboard_url:
               return self.driver.current_url
           else:
               return False
       except Exception as error:
           print("ERROR : ", error)
           return False
~~~

~~~python
"""
Working with dropdown and html tables

https://suman-dynamic-html-form.netlify.app/

# importing the necessary modules
from selenium import webdriver
from selenium.webdriver.chrome.service import Service
from webdriver_manager.chrome import ChromeDriverManager
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import Select
from time import sleep


# Always follow oops concept
class HandleDropDown:

    def __init__(self, url):
        self.url = url
        self.driver = webdriver.Chrome(service=Service(ChromeDriverManager().install()))

    # launch the browser instance
    def start(self):
        self.driver.get(self.url)
        self.driver.maximize_window()
        sleep(5)

    def selectValue(self):
        select = Select(self.driver.find_element(By.ID, 'state'))

        # select by visible text
        # select.select_by_visible_text('Tamilnadu')
        sleep(5)

        # # select by value 
        select.select_by_value('Kerala')

        print('Selected the exact text : Tamilnadu')

    # close all the browser instances
    def shutdown(self):
        self.driver.quit()

if __name__ == "__main__":
    url = "https://suman-dynamic-html-form.netlify.app/"
    myautomation = HandleDropDown(url)
    myautomation.start()
    myautomation.selectValue()
    myautomation.shutdown()
~~~

## Working with Frames

~~~ python
"""
Switching iframe and fetch the content, Drop Here

https://jqueryui.com/droppable/
"""

# importing the necessary modules
from selenium import webdriver
from selenium.webdriver.chrome.service import Service
from webdriver_manager.chrome import ChromeDriverManager
from selenium.webdriver.common.by import By
from time import sleep


# Always follow oops concept
class SwitchFrame:

    def __init__(self, url):
        self.url = url
        self.driver = webdriver.Chrome(service=Service(ChromeDriverManager().install()))

    # launch the browser instance
    def start(self):
        self.driver.get(self.url)
        self.driver.maximize_window()
        sleep(5)

    def fetchText(self):
        frame_element = self.driver.find_element(By.XPATH, "//iframe[@class='demo-frame']")

        self.driver.switch_to.frame(frame_element)
        sleep(2)

        #select the text
        droppable_element = self.driver.find_element(By.XPATH, '//*[@id="droppable"]/p')
        sleep(2)

        print(f'Fetched text : {droppable_element.text}')

    # close all the browser instances
    def shutdown(self):
        self.driver.quit()

if __name__ == "__main__":
    url = "https://jqueryui.com/droppable/"
    myautomation = SwitchFrame(url)
    myautomation.start()
    myautomation.fetchText()
    myautomation.shutdown()
~~~
