##  Practice a Test Case to Write First Script
- Start with a simple user scenario like:
  - Open a browser.
  - Navigate to a website (e.g., https://example.com).
  - Click a button or link.
  - Verify the title or URL.

  
## GUVI Title Verification

  ```python
  from playwright.sync_api import sync_playwright

  def test_example():
      with sync_playwright() as p:
          browser = p.chromium.launch(headless=False)
          page = browser.new_page()
          page.goto("https://www.guvi.in")
          assert page.title() == "GUVI | Learn to Code"
          browser.close()
  ```

##  Assertions
- Assertions are used to verify that a condition is true.
- **Common Playwright Assertions**:
  - Check text content
  - Check element visibility
  - Check attribute values

Example:
```python
"""
Using Playwright Assertions

Assertion 1: Check if the page title is correct
Assertion 2: Check if username field is visible
Assertion 3: Verify input values
Assertion 4: Check the URL for successful navigation
Assertion 5: Verify a specific product is visible
Assertion 6: Check the total number of products

Assertions List: https://playwright.dev/python/docs/test-assertions
"""
from playwright.sync_api import sync_playwright
from playwright.sync_api import expect


def assertion_examples():
    with sync_playwright() as p:
        browser = p.chromium.launch(headless=False)
        page = browser.new_page()

        # Navigate to the Sauce Demo login page
        page.goto("https://www.saucedemo.com/")

        # Assertion 1: Check if the page title is correct
        expect(page).to_have_title("Swag Labs")

        # Assertion 2: Check if username field is visible
        username_field = page.locator("#user-name")
        expect(username_field).to_be_visible()

        # Enter login credentials
        page.fill("#user-name", "standard_user")
        page.fill("#password", "secret_sauce")

        # Assertion 3: Verify input values
        expect(username_field).to_have_value("standard_user")
        password_field = page.locator("#password")
        expect(password_field).to_have_value("secret_sauce")

        # Click the login button
        page.click("#login-button")

        # Assertion 4: Check the URL for successful navigation
        expect(page).to_have_url("https://www.saucedemo.com/inventory.html")

        # Assertion 5: Verify a specific product is visible
        product = page.locator("div.inventory_item_name:has-text('Sauce Labs Backpack')")
        expect(product).to_be_visible()

        # Assertion 6: Check the total number of products
        products = page.locator("div.inventory_item")
        expect(products).to_have_count(6)

        print("All assertions passed successfully!")

        # Close the browser
        browser.close()


if __name__=="__main__":
    assertion_examples()


```

## 📌 Handling Waits in Playwright
- Playwright **auto-waits** for elements to be actionable.
- Explicit waits can be added when needed:
  ```python
  page.wait_for_timeout(3000)  # 3 seconds hard wait (not recommended usually)
  ```
- Prefer **smart waits** like waiting for an element to be attached, visible, etc.

---

##  Wait for Selector and Page Loads
- **Wait for a selector**:
  ```python
  page.wait_for_selector("text=Submit")
  page.click("text=Submit")
  ```
- **Wait for page load**:
  ```python
  page.goto("https://example.com", wait_until="networkidle")
  ```
  `wait_until` options: `'load'`, `'domcontentloaded'`, `'networkidle'`

```python
""""
Explain handling waits with TodoMVC Website
"""
from playwright.sync_api import sync_playwright

def handle_waits():
    with sync_playwright() as p:
        browser = p.chromium.launch(headless=False)
        context = browser.new_context()
        page = context.new_page()

        # Navigate to a sample website
        page.goto("https://demo.playwright.dev/todomvc")

        # Example 1: Using wait_for_selector (Explicit Wait)
        # Wait until a specific element appears
        page.wait_for_selector("h1")  # Wait for the <h1> tag to be visible
        print("Element is visible!")

        # Example 2: Using wait_for_timeout (Static Wait)
        # Wait for 3 seconds
        page.wait_for_timeout(3000)
        print("Waited for 3 seconds!")

        # Example 3: Using wait_until in navigation
        page.goto("https://demo.playwright.dev/todomvc", wait_until="domcontentloaded")  # Wait until all the dom content loaded
        print("DOM Content loaded completely!")

        # Example 4: Using locator.wait_for
        locator = page.locator("text=This is just a demo of TodoMVC for testing, not the ")
        locator.wait_for(timeout=5000)  # Wait until the text "More information" is visible
        print("Text 'demo of TodoMVC' is visible!")

        # Close the browser
        browser.close()

if __name__ == "__main__":
    handle_waits()

```

##  Mini-Project using Microsoft Playwright
### Project: "Login and Dashboard Validation"
- **Objective**: Automate a login and verify dashboard content.
- **Steps**:
  1. Open login page.
  2. Fill username and password.
  3. Click login button.
  4. Wait for dashboard.
  5. Assert successful login by checking welcome text.

## PageObjects/LoginPage
```python
# PageObjects/login_page.py

from playwright.sync_api import sync_playwright, expect

class LoginPage:
    def __init__(self):
        self.playwright = sync_playwright().start()
        self.browser = self.playwright.chromium.launch(headless=False)
        self.context = self.browser.new_context()
        self.page = self.context.new_page()
        self.login_url = "https://opensource-demo.orangehrmlive.com/"

        # Locators
        self.username_input = self.page.locator('[name="username"]') #Attribute (Name)
        self.password_input = self.page.locator('[name="password"]') #Attribute (Name)
        self.login_button = self.page.locator("//button[text()=' Login ']") #xpath
        self.error_message = self.page.locator("//p[text()='Invalid credentials']") #xpath

    def navigate(self):
        self.page.goto(self.login_url)
        expect(self.username_input).to_be_visible()

    def login(self, username: str, password: str):
        self.username_input.fill(username)
        self.password_input.fill(password)
        self.login_button.click()

    def get_error_message(self) -> str:
        expect(self.error_message).to_be_visible()
        return self.error_message.text_content()

    def close(self):
        self.browser.close()
        self.playwright.stop()

```

import pytest
from PageObjects.loginpage import LoginPage
from playwright.sync_api import expect

@pytest.fixture(scope="function")
def login_page():
    page = LoginPage()
    yield page
    # Teardown
    page.close()


class TestOrangeHRMLogin:
    def test_valid_login(self, login_page):
        # Navigate to Login Page
        login_page.navigate()

        # Perform Login with valid credentials
        login_page.login("Admin", "admin123")

        # Assertion: Check if Dashboard is visible by verifying URL contains 'dashboard'
        expect(login_page.page).to_have_url("https://opensource-demo.orangehrmlive.com/web/index.php/dashboard/index")


    def test_invalid_login(self, login_page):
        # Navigate to Login Page
        login_page.navigate()

        # Perform Login with invalid credentials
        login_page.login("InvalidUser", "InvalidPass")

        # Assertion: Check not to have expected URL
        expect(login_page.page).not_to_have_url("https://opensource-demo.orangehrmlive.com/web/index.php/dashboard/index")
