## Python Selenium Exceptions
* Exceptions in Selenium are unexpected events that can occur during test execution and disrupt the normal flow of a program. 
* They can impact the accuracy and reliability of automation.

## Python Selenium Exceptions
* **NoSuchElementException:** Occurs when Selenium cannot find an element on the web page.
* **TimeOutException:** Occurs when a command takes longer to complete than the wait time. (Implicit/ Explicit Waits)
* **ElementNotVisibleException:** Occurs when Selenium tries to locate an element that is hidden on the page.
* **ElementNotInteractableException:** Occurs when a particular HTML element is present in the DOM but you cannot interact with it.
* **ElementClickInterceptedException:** Occurs when an element intercepts the click intended for another element.
* **NoSuchWindowException:** It happens when the intended window is not found.
 

## How to handle exceptions?
* Using **try-except** blocks

## Sample Try-Except Block Code
~~~python
try:
    driver.find_element_by_id("blockedElement").click()

except ElementClickInterceptedException as e:

else:
    print("Unexpected exception!")
finally:
    print("Try except block executed!")
~~~

## 1. Drag and Drop using Action Chains
~~~python
"""
Python Selenium Action Chains : Drag & Drop
"""
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.firefox.service import Service
from webdriver_manager.firefox import GeckoDriverManager
from selenium.common.exceptions import NoSuchElementException
from selenium.common.exceptions import ElementNotVisibleException
from time import sleep


# Action Chains Class
from selenium.webdriver.common.action_chains import ActionChains


class Data:
    url = "https://qavbox.github.io/demo/dragndrop/"


class Locators:
    source = "draggable"
    target = "droppable"


class DragAndDrop(Data, Locators):

    def __init__(self):
        self.driver = webdriver.Firefox(service=Service(GeckoDriverManager().install()))
        self.action = ActionChains(self.driver)


    def drag_drop(self):
        try:
            self.driver.maximize_window()
            self.driver.get(self.url)
            sleep(3)
            source = self.driver.find_element(by=By.ID, value=self.source)
            target = self.driver.find_element(by=By.ID, value=self.target)
            self.action.drag_and_drop(source, target).perform()
            print("Successful Drag and Drop!")
        except (NoSuchElementException, ElementNotVisibleException) as error:
            print(error)
        finally:
            self.driver.quit()


myActions = DragAndDrop()
myActions.drag_drop()
~~~

## 2. Alerts and confirmation boxes using Action Chains
~~~python
"""
Python Selenium Action Chains : Handling Alert Boxes
"""
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.firefox.service import Service
from webdriver_manager.firefox import GeckoDriverManager
from selenium.common.exceptions import NoSuchElementException
from selenium.common.exceptions import ElementNotVisibleException
from selenium.common.exceptions import NoAlertPresentException
from time import sleep


class Data:
    url = "https://qavbox.github.io/demo/alerts/"
    prompt_box_data = "Basil from GUVI"


class Locators:
    alert_box_button = "//input[@name='commit']"
    confirm_box_button = "confirm"
    prompt_box_button = "prompt"


class AlertBox(Data, Locators):

    def __init__(self):
        self.driver = webdriver.Firefox(service=Service(GeckoDriverManager().install()))


    def start(self):
        self.driver.maximize_window()
        self.driver.get(self.url)
        sleep(5)


    def simple_alert_box(self):
        try:
            self.start()
            self.driver.find_element(by=By.XPATH, value=Locators.alert_box_button).click()
            alert_window = self.driver.switch_to.alert
            alert_window.accept()
            print("SUCCESS : OK Button was clicked !")
        except (NoAlertPresentException, NoSuchElementException, ElementNotVisibleException) as error:
            print(error)
        finally:
            self.driver.quit()


    def confirmation_alert_box(self):
        try:
            self.start()
            self.driver.find_element(by=By.ID, value=Locators.confirm_box_button).click()
            alert_window = self.driver.switch_to.alert
            alert_window.dismiss()
            sleep(3)
            print("SUCCESS : Handled confirmation alert box!")
        except (NoAlertPresentException, NoSuchElementException, ElementNotVisibleException) as error:
            print(error)
        finally:
            self.driver.quit()


    def prompt_alert(self):
        try:
            self.start()
            self.driver.find_element(by=By.ID, value=Locators.prompt_box_button).click()
            alert_window = self.driver.switch_to.alert
            alert_window.send_keys(Data.prompt_box_data)
            alert_window.accept()
            sleep(4)
            print("SUCCESS : Handled prompt alert dialog box!")
        except (NoAlertPresentException, NoSuchElementException, ElementNotVisibleException) as error:
            print(error)
        finally:
            self.driver.quit()


suman = AlertBox()
suman.prompt_alert()

# suman.simple_alert_box()
~~~

## Handle Dialog Popup
* Popup represents any ads, form filling, or discounts/offers shown in the website.
  
~~~python
"""
Handle Modal window dialog popup in heroku page
https://the-internet.herokuapp.com/entry_ad
"""
from selenium import webdriver
from selenium.webdriver.chrome.service import Service
from webdriver_manager.chrome import ChromeDriverManager
from selenium.webdriver.common.by import By
from time import sleep

class Data:
    url = "https://the-internet.herokuapp.com/entry_ad"

class Locators:
    modal_locator = '//div[@class="modal"]'  # XPath
    close_button_locator = '//div[@class="modal-footer"]/p' # XPath

class HandlePopup(Data, Locators):
       
    def __init__(self):
        self.driver = webdriver.Chrome(service=Service(ChromeDriverManager().install()))

    def launch_automation(self):
        # Open the demo page
        self.driver.get(Data.url)
        self.driver.maximize_window()
        sleep(5)

        # Click on the button to trigger the alert popup
        if self.driver.find_element(By.XPATH, Locators.modal_locator).is_displayed():
            # Click close button
            self.driver.find_element(By.XPATH, Locators.close_button_locator).click()
            sleep(4)
            print("Popup handled successfully!")
        else:
            print("ERROR")

        # Close the browser
        self.driver.quit()

if __name__ == "__main__":
    mypopup = HandlePopup()
    mypopup.launch_automation()
~~~
